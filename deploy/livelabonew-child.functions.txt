/* 画像ヘルパー：管理画面で設定があればその画像URL、無ければ従来のURLを返す */
if (!function_exists('llabo_img')) {
  function llabo_img($key, $default = '') {
    if (function_exists('get_field')) {
      $v = get_field($key, 'option');
      if (!empty($v)) {
        if (is_array($v) && !empty($v['url'])) return $v['url'];
        if (is_numeric($v)) { $u = wp_get_attachment_image_url($v, 'full'); if ($u) return $u; }
        if (is_string($v)) return $v;
      }
    }
    return $default;
  }
}

/* テキストヘルパー：管理画面で設定があればその表示名、無ければ従来の名前を返す */
if (!function_exists('llabo_text')) {
  function llabo_text($key, $default = '') {
    if (function_exists('get_field')) {
      $v = get_field($key, 'option');
      if (is_string($v) && $v !== '') return $v;
    }
    return $default;
  }
}
